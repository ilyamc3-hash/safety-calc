<script>
let currentType = 'fire';
function setWorkType(type, element) {
currentType = type;
document.querySelectorAll('.type-btn').forEach(b => b.classList.remove('active'));
element.classList.add('active');
document.querySelectorAll('.work-fields').forEach(f => f.classList.add('hidden'));
document.getElementById(`group-${type}`).classList.remove('hidden');
const icons = { fire: '🔥', gas: '💨' };
document.getElementById('center-ico').textContent = icons[type];
calculate();
}
function calculate() {
let zoneMax = 50;
let zoneMin = 20;
let ntd = "";
let z1Name = "Опасная зона";
let z2Name = "Зона ГВС";
if (currentType === 'fire') {
const place = document.getElementById('fire-place').value;
const height = document.getElementById('fire-height').value;
let sparkRadius = 5;
if (height === "2") sparkRadius = 8;
else if (height === "5") sparkRadius = 10;
else if (height === "10") sparkRadius = 12;
else if (height === "15") sparkRadius = 14;
z2Name = "Очистка от материалов";
zoneMin = sparkRadius;
if (place === 'line') {
zoneMax = 50;
z1Name = "Посты оцепления ЛЧ";
ntd = `При работах на линейной части магистральных трубопроводов посты оцепления выставляются на расстоянии <strong>50м</strong>. Очистка площадки от горючих материалов в зависимости от высоты: <strong>${sparkRadius}м</strong>.`;
} else if (place === 'station_open') {
zoneMax = 25;
z1Name = "Опасная зона площадки";
ntd = `На открытых производственных площадках опасная зона составляет не менее <strong>25м</strong>. Заранее очищаемый радиус разлета горячих искр — <strong>${sparkRadius}м</strong>.`;
} else {
zoneMax = 15;
z1Name = "Зона контроля цеха";
ntd = `Внутри замкнутых производственных помещений зона контроля составляет <strong>15м</strong>. Очистка или экранирование горючих конструкций обязательны в радиусе <strong>${sparkRadius}м</strong>.`;
}
}
else if (currentType === 'gas') {
const task = document.getElementById('gas-task').value;
const dn = parseInt(document.getElementById('gas-dn').value);
const p = parseFloat(document.getElementById('gas-pressure').value) || 0;
z1Name = "Зона оцепления";
z2Name = "Анализ воздуха";
if (task === 'purge') {
if (dn >= 1000 && p > 5.4) {
zoneMax = 300;
} else if (dn >= 500 || p > 2.0) {
zoneMax = 200;
} else {
zoneMax = 150;
}
zoneMin = Math.round(zoneMax * 0.25);
ntd = `При интенсивном стравливании или продувке газа через свечи на трубопроводе DN${dn} под давлением ${p} МПа зона оцепления назначается максимальной: <strong>${zoneMax}м</strong>. Контроль шлейфа рассеивания: <strong>${zoneMin}м</strong>.`;
}
else if (task === 'repair_open') {
zoneMax = (p > 5.4 || dn >= 1000) ? 50 : 25;
zoneMin = 15;
ntd = `Ремонтные работы с разгерметизацией полостей оборудования (DN${dn}, давление ${p} МПа). Опасная зона выставления сигнальных постов: <strong>${zoneMax}м</strong>. Радиус непрерывного контроля ГВС: <strong>15м</strong>.`;
}
else {
zoneMax = 15;
zoneMin = 5;
ntd = `Проведение работ без разгерметизации или внутри замкнутых пространств (ОЗП). Минимальная опасная зона составляет <strong>15м</strong>. Анализ воздуха перед входом обязателен в пределах 5м.`;
}
}
// Обновление цифр
document.getElementById('res-d').textContent = zoneMax;
document.getElementById('res-w').textContent = zoneMin;
document.getElementById('ntd-text').innerHTML = ntd;
document.getElementById('lbl-z1').textContent = z1Name;
document.getElementById('lbl-z2').textContent = z2Name;
// Динамическое масштабирование кругов схемы
let maxPx = 200;
let scale = maxPx / zoneMax;
document.getElementById('c1').style.width = `${maxPx}px`;
document.getElementById('c1').style.height = `${maxPx}px`;
// Минимальный размер внутреннего круга 75px, чтобы текст гарантированно помещался ниже эмодзи центра
let innerPx = Math.max(75, zoneMin * scale);
document.getElementById('c2').style.width = `${innerPx}px`;
document.getElementById('c2').style.height = `${innerPx}px`;
}
function triggerAnimate() {
const btn = document.querySelector('.calc-btn');
btn.textContent = '🔄 Расчет...';
setTimeout(() => {
calculate();
btn.textContent = '✅ Готово';
btn.style.background = 'var(--green)';
setTimeout(() => {
btn.textContent = '⚡ Обновить расчет';
btn.style.background = 'var(--orange)';
}, 1200);
}, 400);
}
calculate();
</script>
